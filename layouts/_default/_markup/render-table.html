{{/* Two separate questions about the header row, which are easy to conflate: - does the table HAVE a header? Any content counts, including a cell whose only content is a figure. Markdown always produces a , so a table used purely to align formulas would otherwise draw a header rule under nothing. - what should the table be CALLED? Only readable words count. A cell containing an image shortcode yields markup, which must never reach the aria-label — unescaped it terminates the attribute early. */}} {{- $raw := "" -}} {{- range .THead }}{{ range . }}{{ $raw = printf "%s %s" $raw (.Text | plainify) }}{{ end }}{{ end -}} {{- $hasHead := gt (len (trim $raw " \n\t")) 0 -}} {{- $caption := $raw -}} {{/* Hugo swaps shortcodes for placeholder tokens before Markdown runs and substitutes the rendered markup back afterwards — after this template has finished. A cell containing a figure therefore looks like plain letters here, and its lands in the attribute later, unescaped, terminating it early. The token has to be removed while it is still a token. */}} {{- $caption = replaceRE `HAHAHUGOSHORTCODE\w*` " " $caption -}} {{- $caption = replaceRE `\{\{[<%][^}]*[>%]\}\}` " " $caption -}} {{- $caption = replaceRE `<[^>]*>?` " " $caption -}} {{/* markup, closed or not */}} {{- $caption = replaceRE `[^\pL\pN .,:;()\[\]/+\-–—]+` " " $caption -}} {{- $caption = trim (replaceRE `\s+` " " $caption) " " -}} {{- if gt (len $caption) 80 }}{{ $caption = printf "%s…" (substr $caption 0 80) }}{{ end -}} {{- $n := add (.Ordinal | default 0) 1 -}} {{- $label := cond (gt (len $caption) 0) (printf "Table %d: %s" $n $caption) (printf "Table %d" $n) -}}
