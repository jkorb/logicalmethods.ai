{{/* Hugo's PrevInSection/NextInSection follow the section's default ordering, which runs opposite to chapter order here. Derive them from weight instead so "previous" really is the lower-numbered chapter. */}} {{- $cur := . -}} {{- $pages := where .CurrentSection.Pages.ByWeight "Params.locked" "!=" true -}} {{- $prev := "" -}}{{- $next := "" -}} {{- range $pages -}} {{- if lt .Weight $cur.Weight -}} {{- $prev = . -}} {{- else if and (gt .Weight $cur.Weight) (not $next) -}} {{- $next = . -}} {{- end -}} {{- end -}} {{- $names := .CurrentSection.Param "section-names" -}} {{- if or $prev $next -}}
{{- with $prev }} Previous {{ .Title }} {{- else }}{{ end }} {{- with $next }} Next {{ .Title }} {{- else }}{{ end }}
{{- end -}}
