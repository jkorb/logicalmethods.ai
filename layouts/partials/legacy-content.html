{{/* Compatibility for unrevised chapters during the staged release. */}} {{- $patternMathBlock := `(?s)\$\$(.*?)\$\$` -}} {{- $replacementMathBlock := `
${1}
` -}} {{- $patternMathInline := `(?s)\$(.*?)\$` -}} {{- $replacementMathInline := `${1}` -}} {{- $patternCodeBlock := "```([^`]*)```" -}} {{- $replacementCodeBlock := `
${1}
` -}} {{- $patternCodeInline := "`([^`]*)`" -}} {{- $replacementCodeInline := `${1}` -}} {{- $patternBoolean := "!!(.*?)!!" -}} {{- $replacementBoolean := `${1}` -}} {{- $patternKleene := "~!(.*?)!~" -}} {{- $replacementKleene := `${1}` -}} {{- $patternMDCode := "~~~(.*?)~~~" -}} {{- $replacementMDCode := "```${1}```" -}} {{- $patternMDInline := "%(.*?)%" -}} {{- $replacementMDInline := "`${1}`" -}} {{/* On a LaTeX page the $ delimiters have to survive for KaTeX. */}} {{- $processed := .RawContent -}} {{- if not (.Param "latex") -}} {{- $processed = $processed | replaceRE $patternMathBlock $replacementMathBlock | replaceRE $patternMathInline $replacementMathInline -}} {{- end -}} {{- $processed = $processed | replaceRE $patternCodeBlock $replacementCodeBlock | replaceRE $patternCodeInline $replacementCodeInline | replaceRE $patternBoolean $replacementBoolean | replaceRE $patternKleene $replacementKleene | replaceRE $patternMDCode $replacementMDCode | replaceRE $patternMDInline $replacementMDInline -}} {{ return ($processed | .RenderString) }}
