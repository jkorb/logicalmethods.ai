{{/* One renderer for page-bundle and shared figures, including set members. src is a bundle path, or /img/... relative to assets for shared artwork. */}} {{- $src := .src -}} {{- $res := false -}} {{- if hasPrefix $src "/" -}} {{- $res = resources.Get (strings.TrimPrefix "/" $src) -}} {{- else -}} {{- $res = .page.Resources.GetMatch $src -}} {{- end -}} {{- if not $res -}}{{ errorf "Image not found: %s (%s)" $src .position }}{{- end -}} {{- $decls := slice -}} {{- with .width }}{{ $decls = $decls | append (printf "--figure-width: %s" .) }}{{ end -}} {{- with .height }}{{ $decls = $decls | append (printf "--figure-height: %s" .) }}{{ end -}} {{- with .style }}{{ $decls = $decls | append (strings.TrimRight ";" .) }}{{ end -}} {{- if eq $res.MediaType.Type "image/svg+xml" -}} {{- $svg := $res.Content | replaceRE `<\?xml[^>]*>|]*>` "" -}} {{/* Direct Excalidraw exports work too: normalize root metadata only, never child geometry. The site already supplies the drawing fonts. */}} {{- $svg = replaceRE `(?s)` "" $svg -}} {{- if .themed }}{{ $svg = replaceRE `(stroke|fill)="#(?:000000|1e1e1e|000)"` `${1}="currentColor"` $svg }}{{ $svg = replaceRE `(stroke|fill)="#(?:ffffff|fff)"` `${1}="var(--paper)"` $svg }}{{ end -}} {{- $root := index (findRE `]*>` $svg 1) 0 -}} {{- $normalized := replaceRE `\s(?:width|height|class|role|aria-label|aria-labelledby|aria-hidden|focusable)="[^"]*"` "" $root -}} {{- $access := `aria-hidden="true"` -}} {{- with .alt }}{{ $access = printf `aria-label="%s"` (htmlEscape .) }}{{ end -}} {{- $normalized = replace $normalized " {{- end -}}
