{{- /* Figures prefer a vector sibling. An SVG figure is inlined rather than linked, so it uses the page's own fonts (an cannot) and can pick up currentColor. Excalidraw exports carry no ids or internal references, so inlining several on one page is safe. */ -}} {{- $src := .Get "src" -}} {{- $alt := .Get "alt" | default "" -}} {{- $class := .Get "class" -}} {{- $svg := .Page.Resources.GetMatch (replaceRE `\.(png|jpg|jpeg)$` ".svg" $src) -}} {{- if $svg -}} {{- $markup := $svg.Content -}} {{- $markup = replace $markup `class="figure-svg"` (printf `class="figure-svg %s"` $class) 1 -}} {{- with $alt -}} {{- $markup = replace $markup `role="img"` (printf `role="img" aria-label="%s"` .) 1 -}} {{- else -}} {{- $markup = replace $markup `role="img"` `role="img" aria-hidden="true"` 1 -}} {{- end -}} {{ $markup | safeHTML }} {{- else -}} {{- $img := .Page.Resources.GetMatch $src -}} {{- if $img }} {{ else }} {{ errorf "Image not found: %s (%s)" $src .Position }} {{ end -}} {{- end -}}
