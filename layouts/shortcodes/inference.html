{{/* A Prawitz-style inference figure: premises on a rule, conclusion under it. {{< inference rule="MP" >}} $A → B$ $A$ --- $B$ {{< /inference >}} Lines above the --- are premises, one per line; the content below is the conclusion. Whitespace and blank lines around the separator are ignored. Use layout="stacked" for premises underneath one another. The figure carries a text description, so a screen reader gets the inference as a sentence rather than three loose fragments. */}} {{- $separator := "___HUGO_INFERENCE_SEPARATOR___" -}} {{- $inner := trim .Inner " \t\r\n" -}} {{/* Normalize a --- line, allowing horizontal whitespace around it. Blank lines before and after it remain harmless because each part is trimmed. */}} {{- $inner = replaceRE `(?m)^[ \t]*---[ \t]*\r?$` $separator $inner -}} {{- $parts := split $inner $separator -}} {{- if ne (len $parts) 2 -}} {{- errorf "inference: needs exactly one --- line separating premises from the conclusion (%s)" .Position -}} {{- end -}} {{- $premises := slice -}} {{- range (split (index $parts 0) "\n") -}} {{- with (trim . " \t\r") -}} {{- $premises = $premises | append . -}} {{- end -}} {{- end -}} {{- $conclusion := trim (index $parts 1) " \t\r\n" -}} {{- $plainPremises := apply $premises "replaceRE" "[`$]" "" "." -}} {{- $plain := printf "From %s, infer %s" (delimit $plainPremises ", " ", and ") ($conclusion | replaceRE "[`$]" "" | plainify) -}} {{- partial "figures/inference.html" (dict "premises" $premises "conclusion" $conclusion "plain" $plain "layout" (.Get "layout") "class" (.Get "class") "rule" (.Get "rule") "page" .Page) -}}
