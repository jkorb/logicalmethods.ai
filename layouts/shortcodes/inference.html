{{/* A Prawitz-style inference figure: premises on a rule, conclusion under it. {{< inference rule="MP" >}} $A → B$ $A$ --- $B$ {{< /inference >}} Lines above the --- are premises, one per line; the line below is the conclusion. Use layout="stacked" for premises underneath one another. The figure carries a text description, so a screen reader gets the inference as a sentence rather than three loose fragments. */}} {{- $parts := split (trim .Inner "\n ") "\n---\n" -}} {{- if ne (len $parts) 2 }}{{ errorf "inference: needs one --- line separating premises from the conclusion (%s)" .Position }}{{ end -}} {{- $premises := slice -}} {{- range (split (index $parts 0) "\n") }}{{ with (trim . " ") }}{{ $premises = $premises | append . }}{{ end }}{{ end -}} {{- $conclusion := trim (index $parts 1) " \n" -}} {{- $plain := printf "From %s, infer %s" (delimit (apply $premises "replaceRE" "[`$]" "" ".") ", " ", and ") ($conclusion | replaceRE "[`$]" "" | plainify) -}}
